New Database
posts
create table public.posts (
  id uuid not null default extensions.uuid_generate_v4 (),
  slug text not null,
  title text not null,
  description text null,
  link text not null,
  thumbnail_url text null,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  topic_id uuid null,
  constraint blog_posts_pkey primary key (id),
  constraint blog_posts_link_key unique (link),
  constraint blog_posts_slug_key unique (slug),
  constraint fk_topic foreign KEY (topic_id) references topics (id) on delete set null
) TABLESPACE pg_default;

create index IF not exists idx_posts_search on public.posts using gin (
  to_tsvector(
    'simple'::regconfig,
    (
      (COALESCE(title, ''::text) || ' '::text) || COALESCE(description, ''::text)
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_posts_topic_id on public.posts using btree (topic_id) TABLESPACE pg_default;

changes:
Rename to posts
Remove columns: pub_date, author, original_feed_url, blog_content, tags, and source.
Remove unused index, and create new index for searching keyword interest with its aliases in title and description

interests
create table public.interests (
  id uuid not null default extensions.uuid_generate_v4 (),
  name text not null,
  aliases text[] null,
  topic_id uuid null,
  created_at timestamp with time zone null default now(),
  icon text null,
  constraint interests_pkey primary key (id),
  constraint interests_name_key unique (name),
  constraint interests_topic_id_fkey foreign KEY (topic_id) references topics (id) on delete set null
) TABLESPACE pg_default;

create index IF not exists idx_interests_topic_id on public.interests using btree (topic_id) TABLESPACE pg_default;

changes:
Remove is_public, background_color, text_color
